        WorldStaticModel = DirtBag_Ground,
        Tooltip = Tooltip_TraditionalPreservationSalt_SalineSoil,

    /* The soil and the 10 L of liquid (water or brine) in the bucket that will catch the brine. Notes what went in. */
    item LeachingCharge
    {
        DisplayCategory = Material,
        ItemType = base:normal,
        Weight = 16.0,
        Icon = LeachingCharge,
        StaticModel = Bucket,
        WorldStaticModel = TraditionalPreservationSalt.LeachingChargeGround,
        Tooltip = Tooltip_TraditionalPreservationSalt_LeachingCharge,
    }

    /* Shown as Bucket of Brine: a bucket of brine put on the heat turns into it (BrineUI). A food item, because only food runs the vanilla cooking clock and ReplaceOnCooked.
       Cooking advances heat / 1.5 per minute: about 4 hours on a fire, faster in a hot oven. */
    item BucketBrine
    {
        DisplayCategory = Food,
        ItemType = base:food,
        Weight = 11.0,
        Icon = BucketBrine,
        StaticModel = Bucket,
        WorldStaticModel = TraditionalPreservationSalt.BucketBrineGround,
        CantEat = true,
        CantBeFrozen = true,
        IsCookable = true,
        MinutesToCook = 300,
        MinutesToBurn = 1440,
        ReplaceOnCooked = TraditionalPreservationSalt.BucketSaltCrust,
        FoodType = NoExplicit,
        Tooltip = Tooltip_TraditionalPreservationSalt_Brine,
    }

    item BucketSaltCrust
    {
        DisplayCategory = Material,
        ItemType = base:normal,
        Weight = 1.5,
        Icon = BucketSaltCrust,
        StaticModel = Bucket,
        WorldStaticModel = TraditionalPreservationSalt.BucketSaltCrustGround,
        Tooltip = Tooltip_TraditionalPreservationSalt_SaltCrust,
    }

    /* Shown as Evaporated Salt. The id stays LooseSalt so salt in existing saves survives. Half a vanilla shaker of salt per portion. */
    item LooseSalt
    {
        DisplayCategory = Food,
        EatType = GlugFood,
        ItemType = base:food,
        Weight = 0.1,
        Icon = LooseSalt,

        WorldStaticModel = TraditionalPreservationSalt.EvaporatedSaltPile,

        Spice = true,
        HungerChange = -5.0,

        UnhappyChange = 10,
        FoodType = NoExplicit,
        Tags = base:minoringredient;base:salt,
    }

    /* Ten portions of Evaporated Salt, so it seasons and cures like them. */
    item BagOfSalt
    {
        DisplayCategory = Food,
        EatType = GlugFood,
        ItemType = base:food,
        Weight = 0.8,
        WeightEmpty = 0.1,
        Icon = BagOfSalt,

    /* No salt tag, so a sealed box can't be used as salt. */
    item BoxOfSaltBags
    {
        DisplayCategory = Food,
        ItemType = base:normal,
        Weight = 3.5,
        Icon = Parcel_Medium_Food,
        WorldStaticModel = Parcel_Food_Medium,
        Tags = base:ignorezombiedensity,
        DoubleClickRecipe = TraditionalPreservationSalt.UnpackBoxOfSaltBags,

    /* Any full bucket is the liquid: water, or brine to make stronger. OnTest refuses saturated brine.
       RecordInput keeps a sample of the fluid used, which OnCreate reads. */
    craftRecipe PrepareLeachingCharge
    {
        timedAction = Making,
        time = 120,
        Tags = InHandCraft,
        category = Cooking,
        OnTest = TraditionalPreservationSalt.onTestChargeLiquid,
        OnCreate = TraditionalPreservationSalt.onChargePrepared,
        inputs
        {
            item 1 [TraditionalPreservationSalt.SalineSoilSack] mode:destroy,
            item 1 tags[base:metalbucket] mode:destroy,
            -fluid 10.0 [Water;TaintedWater;Brine] mode:mixture flags[RecordInput],
        }
        outputs
        {
            item 1 TraditionalPreservationSalt.LeachingCharge,
            item 1 Base.EmptySandbag,
        }
    }

    /* 7200 is two hours of game seconds. OnStart and OnUpdate hold that to the game clock,
       so it is true at any Day Length. */
    /* The bucket is the only output: the engine clears the vat overlay once every output is gone. */
    craftRecipe LeachBrine
    {
        time = 7200,
        Tags = LeachingVat,
        category = Cooking,
        overlayStyle = Leaching,
        OnStart = TraditionalPreservationSalt.onLeachStart,
        OnUpdate = TraditionalPreservationSalt.onLeachUpdate,
        OnCreate = TraditionalPreservationSalt.onBrineLeached,
        inputs
        {
            item 1 [TraditionalPreservationSalt.LeachingCharge] mode:destroy,
        }
        outputs
        {
            item 1 TraditionalPreservationSalt.BucketBrine,
        }
    }

    craftRecipe ScrapeSaltCrust
    {
        timedAction = MixingBowl,
        time = 60,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        xpAward = Cooking:10,
        OnCreate = TraditionalPreservationSalt.onCrustScraped,
        inputs
        {
            item 1 tags[base:dullknife;base:sharpknife;base:meatcleaver] mode:keep flags[MayDegradeLight],
            item 1 [TraditionalPreservationSalt.BucketSaltCrust] mode:destroy,
        }
        /* One portion here; OnCreate adds the rest of what the crust's Brine boiled down to. */
        outputs
        {
            item 1 TraditionalPreservationSalt.LooseSalt,
            item 1 Base.BucketEmpty,
        }
    }


    /* genericPacking refuses part-used salt, which would otherwise come back as full portions, and paper bags with items in them. */
    /* Evaporated Salt can be part-used, so OnTest refuses that. A full shaker works in every vanilla recipe. */
    craftRecipe RefillShakerEvaporatedSalt
    {
        timedAction = MixingBowl,
        time = 20,
        OnTest = TraditionalPreservationSalt.onTestFullSalt,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        xpAward = Cooking:3,
        inputs
        {
            item 1 [Base.ShakerEmpty] mode:destroy flags[ItemCount],
            item 2 [TraditionalPreservationSalt.LooseSalt] mode:destroy flags[ItemCount],
        }
        outputs
        {
            item 1 Base.Salt,
        }
    }

    /* A shaker is 10 uses and so is this: 10 of the bag's 50. No ItemCount, so a part-used bag pays from what's left.
       The bag's paper comes back from the performRecipe wrap when this empties it. */
    craftRecipe RefillShakerBagOfSalt
    {
        timedAction = MixingBowl,
        time = 20,
        Tags = InHandCraft;Cooking,
        category = Cooking,
        inputs
        {
            item 1 [Base.ShakerEmpty] mode:destroy flags[ItemCount],
            item 10 [TraditionalPreservationSalt.BagOfSalt],
        }
        outputs
        {
            item 1 Base.Salt,
        }
    }

    craftRecipe CreateBagOfSalt
    {
        timedAction = UnPackSmallBag,
        time = 25,
        OnTest = RecipeCodeOnTest.genericPacking,
        Tags = InHandCraft;Packing;CanBeDoneInDark,
        category = Packing,

        {
            item 10 [TraditionalPreservationSalt.LooseSalt] flags[AllowFavorite;InheritFavorite;ItemCount] mode:destroy,
            item 1 [Base.PaperBag;Base.Paperbag_Jays;Base.Paperbag_Spiffos] mode:destroy,
        }
        outputs
        {
            item 1 TraditionalPreservationSalt.BagOfSalt,
        }
    }

        Tags = InHandCraft;CanBeDoneInDark,
